cmake_minimum_required(VERSION 3.22)
project(NFSUGAndroidCore LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID)
 message(FATAL_ERROR "Use the Android NDK toolchain for this target. Windows equivalence has a separate isolated build.")
endif()
if(NOT NFSUG_MGBA_SOURCE OR NOT EXISTS "${NFSUG_MGBA_SOURCE}/CMakeLists.txt")
 message(FATAL_ERROR "Set NFSUG_MGBA_SOURCE to the existing patched mGBA source/cache.")
endif()
set(LIBMGBA_ONLY ON CACHE BOOL "" FORCE)
set(M_CORE_GBA ON CACHE BOOL "" FORCE)
set(M_CORE_GB OFF CACHE BOOL "" FORCE)
set(USE_DEBUGGERS OFF CACHE BOOL "" FORCE)
set(USE_EDITLINE OFF CACHE BOOL "" FORCE)
set(USE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(BUILD_QT OFF CACHE BOOL "" FORCE)
set(BUILD_SDL OFF CACHE BOOL "" FORCE)
set(BUILD_GL OFF CACHE BOOL "" FORCE)
set(BUILD_GLES2 OFF CACHE BOOL "" FORCE)
set(BUILD_GLES3 OFF CACHE BOOL "" FORCE)
set(BUILD_LIBRETRO OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(COLOR_16_BIT OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
add_subdirectory("${NFSUG_MGBA_SOURCE}" mgba)
add_library(nfsug_android_core SHARED nfsug_android_core.c)
target_compile_definitions(nfsug_android_core PRIVATE BUILD_STATIC M_CORE_GBA)
target_include_directories(nfsug_android_core PRIVATE "${NFSUG_MGBA_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/mgba/include")
target_link_libraries(nfsug_android_core PRIVATE mgba)
target_link_options(nfsug_android_core PRIVATE "-Wl,--no-undefined" "-Wl,-z,max-page-size=16384")

option(NFSUG_ANDROID_JNI "Build the Java core boundary" OFF)
if(NFSUG_ANDROID_JNI)
 target_sources(nfsug_android_core PRIVATE nfsug_android_jni.c)
endif()
