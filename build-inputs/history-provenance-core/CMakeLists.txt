cmake_minimum_required(VERSION 3.22)
project(NFSUGAndroidCapture LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(NOT ANDROID)
 message(FATAL_ERROR "NDK toolchain required.")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/capture_sources.cmake")
add_library(nfsug_android_capture SHARED nfsug_android_core.c observer.c native_span.c ${NF_CAPTURE_SOURCES})
target_include_directories(nfsug_android_capture PRIVATE "." camera "${NF_MGBA_SOURCE}/include" "${NF_MGBA_BUILD}/mgba/include")
set(NF_VISIBILITY_QUEUE_CAPACITY 20 CACHE STRING "Owned visibility queue ABI capacity (20 legacy or 64 wide)")
if(NOT NF_VISIBILITY_QUEUE_CAPACITY MATCHES "^(20|64)$")
 message(FATAL_ERROR "Visibility queue capacity must be 20 or 64")
endif()
target_compile_definitions(nfsug_android_capture PRIVATE BUILD_STATIC M_CORE_GBA NF_VISIBILITY_CAPTURE NF_ROAD_ATTRIBUTE_CORRECTION NF_PACKET_NO_FILES _GNU_SOURCE NF_VISIBILITY_QUEUE_CAPACITY=${NF_VISIBILITY_QUEUE_CAPACITY})
target_compile_options(nfsug_android_capture PRIVATE -fvisibility=hidden -ffunction-sections -fdata-sections -include "${CMAKE_CURRENT_SOURCE_DIR}/windows.h")
target_link_libraries(nfsug_android_capture PRIVATE "${NF_MGBA_BUILD}/mgba/libmgba.a" m atomic)
target_link_options(nfsug_android_capture PRIVATE "-Wl,--no-undefined" "-Wl,--gc-sections" "-Wl,-z,max-page-size=16384")

option(NF_CAPTURE_JNI "Build app JNI into the same core image" OFF)
if(NF_CAPTURE_JNI)
 target_sources(nfsug_android_capture PRIVATE nfsug_android_jni.c)
 target_compile_definitions(nfsug_android_capture PRIVATE NFA_WITH_NATIVE_CAPTURE)
 set_target_properties(nfsug_android_capture PROPERTIES OUTPUT_NAME nfsug_android_core)
endif()
